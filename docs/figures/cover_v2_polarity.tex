\begin{scope}[shift={(15.0,0)}]
\node[title] at (0,6.05) {(c) Edge projection $\pi$ (when $\rho_i=\rho_j$)};
\node[anchor=north west, font=\scriptsize] at (0,5.85) {%
\renewcommand{\arraystretch}{1.25}%
\begin{tabular}{@{}l c c@{}}
lexical edge $p_i \to h_j$ & context \up & context \dn \\ \hline

hyponym, \textit{dog}$\to$\textit{animal} ($\sqsubset$) & \textcolor{logic}{$\sqsubset$ supports} & \textcolor{bad}{$\sqsupset$ does not} \\
hypernym, \textit{animal}$\to$\textit{dog} ($\sqsupset$) & \textcolor{bad}{$\sqsupset$ does not} & \textcolor{logic}{$\sqsubset$ supports} \\
antonym ($\,|\,$) & $|$ & $|$ \;{\tiny(conflict head)} \\
\end{tabular}};
\node[anchor=west, font=\tiny, text=mute, text width=5.6cm, align=left] at (0,3.3)

\end{scope}
